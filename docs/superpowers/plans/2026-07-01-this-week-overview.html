Implementation Plan · henceforth-club · branch feat/this-week-overview

This Week in Parliament — the one-page overview

The plain-English, single-A4 overview becomes the standard weekly page. The long-form article moves to a linked /full sub-route. The overview's prose lives in a new overview block on the digest that the generator writes.

Implements docs/superpowers/specs/2026-07-01-this-week-overview-design.html (approved 2026-07-01). Execute task-by-task with superpowers:subagent-driven-development.

For agentic workers: REQUIRED SUB-SKILL — superpowers:subagent-driven-development. Steps are real checkboxes. Pure helpers are TDD: write the failing vitest first, watch it fail, then implement. React components are verified by the screenshot pass in Task 15, not unit tests. Commit after each task. A verified reference renderer already exists (produced this design session); its exact CSS and logic are reproduced in the tasks below, so nothing is invented.

Global constraints

  • Next.js App Router, TypeScript, React 19. Server components by default; only A4Sheet is a client component ('use client').
  • Package manager: match the repo (package.json uses vitest run for npm test). ES-module scripts under scripts/this-week are .mjs.
  • Styling: one CSS module, overview.module.css, co-located with the components. Ported verbatim from the verified renderer. Point sizes and the A4 geometry are load-bearing — do not "tidy" them.
  • Print: browser print only. @page { size: A4 } plus the client auto-fit that shrinks the base font until the content fits 297mm.
  • Prose rules for the overview block (enforced by Task 11's lint): no em-dashes or connector colons; commas sparingly; whole numbers up to ninety-nine spelled; numerals for vote tallies, dates, percentages and large figures; proper terms for role and party initialisms (Members of Parliament, Prime Minister), established proper nouns (NHS) kept; every fact and quotation preserved.
  • Git: this branch, feat/this-week-overview. Feature branch, never main (Vercel auto-deploys main). No force-push. Prose in commits and docs spells out acronyms.
  • Definition of done includes a screenshot pass (Task 15). Render, screenshot with headless Chrome, read the image, before declaring a visual task complete.

    import type { DivisionRow } from './types'

export interface PreparedDivision {
  title: string; day: number; ayes: number; noes: number
  carried: boolean; margin: number; ayesPct: number
}
export interface PreparedDivisions {
  mode: 'none' | 'pills' | 'rows'
  items: PreparedDivision[]
  overflow: number          // count collapsed into the summary line
  overflowAllCarried: boolean
}

export function shortenDivTitle(t: string): string {
  return t.replace(/^Draft /, '').replace(/ 2026$/, '')
    .replace(/Customs \(Tariff and Miscellaneous Amendments\) \(No\. \d+\) Regulations/, 'Customs (Tariff) Regulations')
    .replace(/Climate Change Act 2008 \(International Aviation and International Shipping\) Regulations/, 'Climate Change Act, Aviation and Shipping')
    .replace(/Climate Change Act 2008 \(Credit Limit\) Order/, 'Climate Change Act, Credit Limit')
    .replace(/National Security \(State Threats\) Bill(?: Committee)? — /, 'National Security Bill, ')
    .replace(/National Security \(State Threats\) Bill(?: Committee)?/, 'National Security Bill')
    .replace(/Armed Forces Bill Report Stage — /, 'Armed Forces Bill, ')
    .replace(/Opposition Day — /, 'Opposition Day, ')
    .replace(/ — /g, ', ')
}

export function shortenDept(s: string): string {
  return s
    .replace('Department of Health and Social Care', 'Health').replace('Ministry of Defence', 'Defence')
    .replace('Ministry of Housing, Communities and Local Government', 'Housing').replace('Department for Transport', 'Transport')
    .replace('Department for Environment, Food and Rural Affairs', 'Environment').replace('Department for Education', 'Education')
    .replace('Department for Business and Trade', 'Business and Trade').replace('Department for Energy Security and Net Zero', 'Energy')
    .replace('Department for Work and Pensions', 'Work and Pensions')
    .replace('Foreign, Commonwealth and Development Office', 'Foreign Office')
    .replace(/^(\d+) other departments$/, '$1 others')
}

const CAP = 16
export function prepareDivisions(votes: { row: DivisionRow }[]): PreparedDivisions {
  if (!votes.length) return { mode: 'none', items: [], overflow: 0, overflowAllCarried: true }
  const all = votes.map(v => {
    const ayes = v.row.ayes, noes = v.row.noes, tot = ayes + noes || 1
    return { title: shortenDivTitle(v.row.title), day: parseInt(v.row.date.slice(8), 10),
      ayes, noes, carried: ayes > noes, margin: Math.abs(ayes - noes), ayesPct: ayes / tot * 100 }
  }).sort((a, b) => a.margin - b.margin)
  if (all.length <= 6) return { mode: 'pills', items: all, overflow: 0, overflowAllCarried: true }
  const items = all.slice(0, CAP), rest = all.slice(CAP)
  return { mode: 'rows', items, overflow: rest.length, overflowAllCarried: rest.every(r => r.carried) }
}

export function trimSentences(text: string, n: number): string {
  const parts = String(text || '').split(/(?<=\.) /)
  const bal = (s: string) => (s.match(/\(/g) || []).length - (s.match(/\)/g) || []).length
  let out = parts.slice(0, n).join(' ')
  for (let i = n; bal(out) > 0 && i < parts.length && i < n + 3; i++) out += ' ' + parts[i]
  if (bal(out) > 0) out = out.slice(0, out.lastIndexOf('(')).trim()
  return out
}

import { useEffect, useRef } from 'react'
import s from './overview.module.css'

export default function A4Sheet({ children }: { children: React.ReactNode }) {
  const ref = useRef<HTMLDivElement>(null)
  useEffect(() => {
    const el = ref.current; if (!el) return
    const A4 = 297 * 96 / 25.4
    let fs = 10.5; el.style.fontSize = fs + 'pt'
    for (let g = 0; el.scrollHeight > A4 && fs > 7 && g < 50; g++) { fs -= 0.2; el.style.fontSize = fs + 'pt' }
  }, [])
  return (
    <>
      <button className={s.printBtn + ' noprint'} onClick={() => window.print()}>Print</button>
      <div className={s.sheet} ref={ref}>{children}</div>
    </>
  )
}
